func mat(color: Color, glow: float = 0.0) -> StandardMaterial3D:
	var m = StandardMaterial3D.new()
	m.albedo_color = color
	m.roughness = 0.8
	if glow > 0:
		m.emission_enabled = true
		m.emission = color
		m.emission_energy_multiplier = glow
	return m

func box(parent: Node3D, pos: Vector3, size: Vector3, color: Color) -> MeshInstance3D:
	var n = MeshInstance3D.new()
	var mesh = BoxMesh.new()
	mesh.size = size
	n.mesh = mesh
	n.material_override = mat(color)
	n.position = pos
	parent.add_child(n)
	return n

func sphere(parent: Node3D, pos: Vector3, radius: float, color: Color) -> MeshInstance3D:
	var n = MeshInstance3D.new()
	var mesh = SphereMesh.new()
	mesh.radius = radius
	mesh.height = radius * 2
	n.mesh = mesh
	n.material_override = mat(color)
	n.position = pos
	parent.add_child(n)
	return n

func cylinder(parent: Node3D, pos: Vector3, radius: float, height: float, color: Color) -> MeshInstance3D:
	var n = MeshInstance3D.new()
	var mesh = CylinderMesh.new()
	mesh.top_radius = radius
	mesh.bottom_radius = radius
	mesh.height = height
	n.mesh = mesh
	n.material_override = mat(color)
	n.position = pos
	parent.add_child(n)
	return n

func person(pos: Vector3, color: Color, is_player: bool = false) -> Node3D:

func upgrade_cost() -> int:
	return 100 + level*100

func buy_upgrade():
	if level<4 and money>=upgrade_cost():
		money-=upgrade_cost()
		level+=1
		save_game()
		show_prep("厨房升级完成！")

func spawn_customer():
	var available = []
	for i in range(tables.size()):
		if tables[i].state == "empty": available.append(i)
	if available.is_empty(): return
	var t = tables[available[rng.randi_range(0,available.size()-1)]]
	t.state = "waiting"
	t.food = rng.randi_range(0,2)
	t.patience = 57.0 + level*5
	t.guest.show()

func toast(message: String):
	hint.text = message
	toast_time = 5

func interact(index: int):
	if not active or paused: return
	if index == 0:
		if selected == 2: toast("饮料请在右侧茶水台制作。")
		elif cooking>=0 or ready_food>=0: toast("厨房已有料理，先去出餐口取餐。")
		else:
			cooking = selected
			cook_time = maxf(1.8,4.8-level*0.7)
			toast("师傅正在制作 "+FOOD[selected]+"，可以先去清桌。")
	elif index == 1:
		if carrying>=0: toast("托盘已有餐品，请先送餐。")
		elif ready_food>=0:
			carrying = ready_food
			ready_food = -1
			toast("已取 "+FOOD[carrying]+"，送到相应客人桌边。")
		else: toast("餐品还未做好。先到料理台下单。")
	elif index == 2:
		if carrying<0:
			carrying = 2
			toast("已拿取柚子气泡饮。")
		else: toast("托盘已满。按 Q 放弃手中餐品。")
	else:
		var t = tables[index-3]
		if t.state == "dirty":
			t.state = "empty"
			t.dish.hide()
			toast("桌面清理完毕，准备迎接下一位客人。")
		elif t.state == "waiting":
			if carrying == t.food:
				var tip = int(t.patience/10)+mini(combo,5)*2
				var revenue = PRICE[carrying]+tip
				earnings += revenue
				money += revenue
				served += 1
				combo += 1
				carrying = -1
				t.state = "eating"
				t.timer = 7
				t.dish.show()
				toast("好评！收入 ¥%d（小费 ¥%d） · 连续服务 %d" % [revenue,tip,combo])
			else: toast("这位客人要的是 "+FOOD[t.food]+"。")
		elif t.state == "empty": toast("空桌，稍后会有客人入座。")

func _unhandled_input(event):
	if event is InputEventKey and event.pressed and not event.echo:
		if event.keycode == KEY_ESCAPE and active:
			paused = not paused
			modal.visible = paused
			if paused:
				modal_text.text = "营业已暂停\n\nWASD / 方向键：移动\nE：与附近设施或餐桌互动\n1 / 2 / 3：选择菜品\nQ：放弃托盘中的餐品\n\n也可以点击场景中的标签自动前往。"
				start_button.text = "继续营业"
				upgrade_button.hide()
		if not active or paused: return
		if event.keycode in [KEY_1,KEY_2,KEY_3]: selected = event.keycode-KEY_1
		if event.keycode == KEY_Q: carrying = -1; toast("托盘已清空。")
		if event.keycode == KEY_E:
			var nearest = -1
			var distance = 1.3
			for i in range(stations.size()):
				var d = player.position.distance_to(stations[i])
				if d<distance: nearest=i; distance=d
			if nearest>=0: interact(nearest)
			else: toast("请走近料理台、出餐口、茶水台或餐桌。")
	if event is InputEventMouseButton and event.pressed and event.button_index == MOUSE_BUTTON_LEFT and active and not paused:
		var origin = camera.project_ray_origin(event.position)
		var direction = camera.project_ray_normal(event.position)
		var p = Plane(Vector3.UP,0).intersects_ray(origin,direction)
		if p != null:
			var nearest = -1
			var distance = 1.8
			for i in range(stations.size()):
				var d = Vector2(p.x,p.z).distance_to(Vector2(stations[i].x,stations[i].z))
				if d<distance: distance=d; nearest=i
			target=nearest

func can_walk(p: Vector3) -> bool:
	if abs(p.x)>7.7 or p.z < -2.6 or p.z>5.4: return false
	for t in tables:
		if Vector2(p.x,p.z).distance_to(Vector2(t.pos.x,t.pos.z))<1.0: return false
	return true

func move_player(delta: float):
	var direction = Vector3.ZERO
	if Input.is_physical_key_pressed(KEY_W) or Input.is_physical_key_pressed(KEY_UP): direction.z-=1
	if Input.is_physical_key_pressed(KEY_S) or Input.is_physical_key_pressed(KEY_DOWN): direction.z+=1
	if Input.is_physical_key_pressed(KEY_A) or Input.is_physical_key_pressed(KEY_LEFT): direction.x-=1
	if Input.is_physical_key_pressed(KEY_D) or Input.is_physical_key_pressed(KEY_RIGHT): direction.x+=1
	if direction.length()>0:
		target=-1
		direction = direction.rotated(Vector3.UP,0.55)
	elif target>=0:
		if route_target != target:
			route_target = target
			var from_cell = Vector2i(roundi(player.position.x*4),roundi(player.position.z*4))
			var to_cell = Vector2i(roundi(stations[target].x*4),roundi(stations[target].z*4))
			route = navigation.get_point_path(from_cell,to_cell)
		if not route.is_empty():
			var waypoint = Vector3(route[0].x,0,route[0].y)
			if player.position.distance_to(waypoint)<0.13:
				route.remove_at(0)
			else: direction=(waypoint-player.position).normalized()
		else:
			if player.position.distance_to(stations[target])<0.6: interact(target)
			target=-1
			route_target=-1
	if target<0: route_target=-1
	if direction.length()>0:
		direction=direction.normalized()
		var step=direction*delta*4.6
		var p=player.position
		if can_walk(p+Vector3(step.x,0,0)): p.x+=step.x
		if can_walk(p+Vector3(0,0,step.z)): p.z+=step.z
		player.position=p
		player.rotation.y=lerp_angle(player.rotation.y,atan2(direction.x,direction.z),delta*14)
		for i in range(legs.size()): legs[i].rotation.x=sin(time*16+i*PI)*0.35
	else:
		for leg in legs: leg.rotation.x=0

func _process(delta):
	time+=delta
	if active and not paused:
		remaining=maxf(0,remaining-delta)
		move_player(delta)
		spawn_clock-=delta
		if spawn_clock<=0:
			spawn_customer()
			spawn_clock=maxf(7,13-night)
		if cooking>=0:
			cook_time-=delta
			chef.rotation.y=sin(time*7)*0.25
			if cook_time<=0:
				ready_food=cooking
				cooking=-1
				toast("料理完成！去出餐口拿取 "+FOOD[ready_food]+"。")
		for t in tables:
			if t.state == "waiting":
				t.patience-=delta
				if t.patience<=0:
					t.state="dirty"
					t.guest.hide()
					t.dish.show()
					missed+=1
					combo=0
					toast("客人等得太久离开了，清桌后继续加油。")
			elif t.state == "eating":
				t.timer-=delta
				if t.timer<=0:
					t.state="dirty"
					t.guest.hide()
		if remaining<=0: finish_night()
	plate.visible=carrying>=0
	toast_time-=delta
	if toast_time<=0: hint.text="点击标签自动前往并互动  /  WASD 移动  /  E 互动  /  1–3 选菜  /  Q 清空托盘  /  ESC 暂停"
	update_ui()

func finish_night():
	active=false
	target=-1
	best=maxi(best,earnings)
	var summary="营业结束  ·  服务 %d 位 / 离开 %d 位\n本夜收入 ¥%d   ·   最佳 ¥%d" % [served,missed,earnings,best]
	night+=1
	save_game()
	show_prep(summary)

func update_ui():
	if hud==null: return
	hud.text="第 %02d 夜      %02d:%02d      钱包 ¥%d\n已服务 %d 位    ·    连续好评 %d    ·    本夜 ¥%d" % [night,int(remaining)/60,int(remaining)%60,money,served,combo,earnings]
	tray_label.text="选菜  %s     |     托盘  %s     |     厨房  %s" % [FOOD[selected],FOOD[carrying] if carrying>=0 else "空",("制作中 %.1fs"%maxf(cook_time,0)) if cooking>=0 else ("可取餐" if ready_food>=0 else "空闲")]
	for i in range(bubbles.size()):
		var b=bubbles[i]
		var world=stations[i]+Vector3(0,1.6,-0.3)
		if i<3: world=stations[i]+Vector3(0,1.45,-0.65)
		b.position=camera.unproject_position(world)-Vector2(80,25)
		b.visible=active and not paused
		if i==0: b.text="料理台 · 制作" if cooking<0 else "制作中 %.1fs"%maxf(0,cook_time)
		elif i==1: b.text="取餐："+FOOD[ready_food] if ready_food>=0 else "出餐口"
		elif i==2: b.text="柚子饮 · 拿取"
		else:
			var t=tables[i-3]
			match t.state:
				"empty": b.text="%d 号桌 · 待客"%(i-2)
				"waiting": b.text=FOOD[t.food]+"  %ds"%int(t.patience)
				"eating": b.text="享用中  ♡"
				"dirty": b.text="清理餐桌  ✧"
			b.modulate=Color("ffb4a1") if t.state=="waiting" and t.patience<18 else Color.WHITE

func save_game():
	var f=FileAccess.open(save_path,FileAccess.WRITE)
	if f: f.store_string(JSON.stringify({"money":money,"night":night,"level":level,"best":best}))

func load_game():
	if not FileAccess.file_exists(save_path): return
	var data=JSON.parse_string(FileAccess.get_file_as_string(save_path))
	if data is Dictionary:
		money=int(data.get("money",0))
		night=int(data.get("night",1))
		level=int(data.get("level",0))
		best=int(data.get("best",0))
